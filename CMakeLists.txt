cmake_minimum_required(VERSION 3.16)
project(document_search_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(document_search dict.cpp search.cpp Node.cpp)
target_include_directories(document_search PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(search_demo examples/demo.cpp)
target_link_libraries(search_demo PRIVATE document_search)
enable_testing()
add_executable(search_tests tests/test_engine.cpp)
target_link_libraries(search_tests PRIVATE document_search)
add_test(NAME engine_regression COMMAND search_tests)
